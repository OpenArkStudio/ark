# Declarations stay available to core consumers even without a compiled reader.
add_library(ark_config_headers INTERFACE)
add_library(ark::config_headers ALIAS ark_config_headers)
set(ark_config_include_root "${CMAKE_CURRENT_BINARY_DIR}/public-include")
file(MAKE_DIRECTORY "${ark_config_include_root}/ark")
configure_file(application_manifest.hpp "${ark_config_include_root}/ark/application_manifest.hpp" COPYONLY)
# This module owns this mapping directory; remove obsolete mapped headers.
file(GLOB_RECURSE ark_config_old_headers LIST_DIRECTORIES FALSE "${ark_config_include_root}/ark/*")
foreach(header IN LISTS ark_config_old_headers)
    if(NOT header STREQUAL "${ark_config_include_root}/ark/application_manifest.hpp")
        file(REMOVE "${header}")
    endif()
endforeach()
target_include_directories(ark_config_headers INTERFACE
    "$<BUILD_INTERFACE:${ark_config_include_root}>"
    "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
if(ARK_INSTALL)
    install(FILES application_manifest.hpp DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/ark")
endif()
target_compile_features(ark_config_headers INTERFACE cxx_std_20)

if(ARK_BUILD_CONFIG OR ARK_BUILD_NETWORK_TESTS)
    include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/ark_toml.cmake")
    add_library(ark_config STATIC application_manifest.cpp configuration_reader.cpp)
    add_library(ark::config ALIAS ark_config)
    set_target_properties(ark_config PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_link_libraries(ark_config PUBLIC ark::config_headers)
    get_target_property(toml_target_type tomlplusplus::tomlplusplus TYPE)
    if(toml_target_type STREQUAL "INTERFACE_LIBRARY")
        # Header-only parser code is compiled into the reader, with no link ABI.
        target_link_libraries(ark_config PRIVATE "$<BUILD_INTERFACE:tomlplusplus::tomlplusplus>")
    else()
        target_link_libraries(ark_config PRIVATE tomlplusplus::tomlplusplus)
        set_target_properties(ark_config PROPERTIES ARK_PACKAGE_NEEDS_TOML ON)
    endif()
endif()
